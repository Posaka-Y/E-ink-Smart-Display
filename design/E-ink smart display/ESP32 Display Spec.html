キー型説明
tsnumberUnix timestamp（NTP同期チェック用）
evarrayイベント配列（最大3件）
ev[].dstring日付 MM/DD 形式
ev[].tstring開始時刻 HH:MM（終日は "--"）
ev[].nstringイベント名（最大10文字推奨）
キー名を1文字に短縮して省メモリ化。 ArduinoJson でのパースも最小限のメモリで済みます。 イベントは今日から7日以内の直近3件を返すようにプロキシ側でフィルタします。
Google Apps Script — カレンダープロキシ
Code.gs Google Apps Script 
function doGet(e) {
  const cal  = CalendarApp.getDefaultCalendar();
  const now  = new Date();
  const end  = new Date(now.getTime() + 7 * 86400000);
  const evts = cal.getEvents(now, end).slice(0, 3);

  const pad  = n => String(n).padStart(2, '0');
  const fmtD = d => `${pad(d.getMonth()+1)}/${pad(d.getDate())}`;
  const fmtT = d => d.isAllDayEvent()
    ? '--'
    : `${pad(d.getStartTime().getHours())}:${pad(d.getStartTime().getMinutes())}`;

  const payload = {
    ts: Math.floor(now.getTime() / 1000),
    ev: evts.map(ev => ({
      d: fmtD(ev.getStartTime()),
      t: ev.isAllDayEvent() ? '--'
           : `${pad(ev.getStartTime().getHours())}:${pad(ev.getStartTime().getMinutes())}`,
      n: ev.getTitle().substring(0, 10)
    }))
  };

  return ContentService
    .createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}
デプロイ手順： Apps Script エディタ → デプロイ → 新しいデプロイ → 種類「ウェブアプリ」→ アクセス「全員」→ URLをコピーしてESP32に書く。
ESP32 — Arduino スニペット
main.ino（抜粋） C++ / Arduino 
// ライブラリ: ArduinoJson, HTTPClient, Adafruit_BME280

#include <ArduinoJson.h>
#include <HTTPClient.h>
#include <Adafruit_BME280.h>

const char* GAS_URL = "https://script.google.com/macros/s/YOUR_ID/exec";

struct Event {
  char d[6];   // "04/28"
  char t[6];   // "10:00"
  char n[12];  // イベント名
};

void fetchCalendar(Event* events, uint8_t* count) {
  HTTPClient http;
  http.begin(GAS_URL);
  http.setFollowRedirects(HTTPC_STRICT_FOLLOW_REDIRECTS);

  int code = http.GET();
  if (code != 200) { *count = 0; return; }

  StaticJsonDocument<512> doc;
  deserializeJson(doc, http.getStream());

  JsonArray ev = doc["ev"];
  *count = min((int)ev.size(), 3);
  for (int i = 0; i < *count; i++) {
    strlcpy(events[i].d, ev[i]["d"] | "", sizeof(events[i].d));
    strlcpy(events[i].t, ev[i]["t"] | "", sizeof(events[i].t));
    strlcpy(events[i].n, ev[i]["n"] | "", sizeof(events[i].n));
  }
  http.end();
}

void loop() {
  Event   events[3];
  uint8_t evCount;

  // センサー読み取り（直結）
  float temp = bme.readTemperature();
  float hum  = bme.readHumidity();
  float pres = bme.readPressure() / 100.0F;

  // カレンダー取得
  fetchCalendar(events, &evCount);

  // E-ink 描画（ここにWaveshare描画コード）
  drawDisplay(temp, hum, pres, events, evCount);

  delay(5 * 60 * 1000); // 5分スリープ
}
メモリ節約： StaticJsonDocument<512> で十分。 GASのリダイレクト（302→307）に対応するため setFollowRedirects(HTTPC_STRICT_FOLLOW_REDIRECTS) が必要です。
ESP32 — drawDisplay() 描画コード
前提ライブラリ： Waveshare の EPD_5in79.h / GUI_Paint.h を lib/ に配置。フォントは Font24（英数）と 日本語ビットマップフォント Font_JP16（自前 or u8g2 変換）を使用。 日本語文字列は UTF-8 → Shift-JIS 変換不要な場合は英語表記で代替可。
display.h — バッファ定義 C++ 
// 792×272 の1bppバッファ × 2面（Black / Red）
#define EPD_W  792
#define EPD_H  272
#define BUF_SZ (EPD_W * EPD_H / 8)   // 26928 bytes

static uint8_t imgBlack[BUF_SZ];
static uint8_t imgRed  [BUF_SZ];
display.cpp — drawDisplay() C++ / Arduino 
#include "EPD_5in79.h"
#include "GUI_Paint.h"
#include "Fonts/Font20.h"   // 英数 20px
#include "Fonts/Font64.h"   // 時計用 大フォント
#include "display.h"

// ── レイアウト定数 ─────────────────────────────────────────
#define COL_DIV    340   // 左ペイン幅
#define SENSOR_H   100   // センサー行の高さ
#define PAD        16    // 内側余白

void drawDisplay(
    float   temp,
    float   hum,
    float   pres,
    Event*  events,
    uint8_t evCount)
{
  // ── 初期化 ──────────────────────────────────────────────
  Paint paintB, paintR;
  Paint_NewImage(&paintB, imgBlack, EPD_W, EPD_H, ROTATE_0);
  Paint_NewImage(&paintR, imgRed,   EPD_W, EPD_H, ROTATE_0);
  Paint_SelectImage(&paintB); Paint_Clear(WHITE);
  Paint_SelectImage(&paintR); Paint_Clear(WHITE);

  // ── 枠線 ────────────────────────────────────────────────
  Paint_SelectImage(&paintB);
  // 外枠
  Paint_DrawRectangle(0, 0, EPD_W-1, EPD_H-1, BLACK, DOT_PIXEL_2X2, DRAW_FILL_EMPTY);
  // 左右分割線
  Paint_DrawLine(COL_DIV, 0, COL_DIV, EPD_H, BLACK, DOT_PIXEL_2X2, LINE_STYLE_SOLID);
  // センサー/カレンダー分割線（右ペイン）
  Paint_DrawLine(COL_DIV, SENSOR_H, EPD_W, SENSOR_H, BLACK, DOT_PIXEL_2X2, LINE_STYLE_SOLID);

  // ── 左ペイン: 時計 ───────────────────────────────────────
  char buf[16];
  // 月表示（小さめ）
  sprintf(buf, "2026/04");
  Paint_DrawString_EN(PAD, 14, buf, &Font20, WHITE, BLACK);

  // 曜日バッジ（赤背景）
  Paint_SelectImage(&paintR);
  Paint_DrawRectangle(240, 10, 330, 34, RED, DOT_PIXEL_1X1, DRAW_FILL_FULL);
  Paint_SelectImage(&paintB);
  Paint_DrawString_EN(252, 14, "TUE", &Font20, RED, WHITE);
  // ↑ 日本語フォントがあれば "火" を描画

  // 時刻（大フォント・赤）
  getTimeStr(buf);  // "14:32" を取得するユーティリティ
  Paint_SelectImage(&paintR);
  Paint_DrawString_EN(PAD, 50, buf, &Font64, WHITE, RED);

  // 日付（中サイズ）
  Paint_SelectImage(&paintB);
  sprintf(buf, "28 / 04");
  Paint_DrawString_EN(PAD, 188, buf, &Font20, WHITE, BLACK);

  // ── 右上: センサー ──────────────────────────────────────
  const int sx[3] = {COL_DIV+10, COL_DIV+155, COL_DIV+305};
  const int sw     = 148;
  // 縦区切り線
  Paint_DrawLine(sx[1]-3, 0, sx[1]-3, SENSOR_H, BLACK, DOT_PIXEL_1X1, LINE_STYLE_SOLID);
  Paint_DrawLine(sx[2]-3, 0, sx[2]-3, SENSOR_H, BLACK, DOT_PIXEL_1X1, LINE_STYLE_SOLID);

  // ラベル行
  Paint_DrawString_EN(sx[0]+8, 12, "TEMP", &Font20, WHITE, BLACK);
  Paint_DrawString_EN(sx[1]+8, 12, "HUM",  &Font20, WHITE, BLACK);
  Paint_DrawString_EN(sx[2]+8, 12, "PRES", &Font20, WHITE, BLACK);

  // 数値行（大きめ）
  sprintf(buf, "%.1fC", temp);
  Paint_DrawString_EN(sx[0]+4, 46, buf, &Font20, WHITE, BLACK);
  sprintf(buf, "%.0f%%", hum);
  Paint_DrawString_EN(sx[1]+4, 46, buf, &Font20, WHITE, BLACK);
  sprintf(buf, "%.0fhPa", pres);
  Paint_DrawString_EN(sx[2]+4, 46, buf, &Font20, WHITE, BLACK);

  // ── 右下: 週間カレンダー + イベント ─────────────────────
  const int CAL_TOP  = SENSOR_H + 6;
  const int CAL_H    = 70;   // 曜日セル高さ
  const int EVT_TOP  = CAL_TOP + CAL_H + 4;
  const int CELL_W   = (int)((EPD_W - COL_DIV - 4) / 7);

  // 曜日セル
  const char* dows[] = {"Mo","Tu","We","Th","Fr","Sa","Su"};
  const int   days[] = {28,29,30,1,2,3,4};
  const int   todayIdx = 0;  // 今日 = 月曜（配列index）

  for (int i = 0; i < 7; i++) {
    int cx = COL_DIV + 2 + i * CELL_W;
    bool isToday  = (i == todayIdx);
    bool isSun    = (i == 6);
    bool hasEvent = (i==0||i==1||i==4);

    if (isToday) {
      // 今日セル: 赤枠
      Paint_SelectImage(&paintR);
      Paint_DrawRectangle(cx, CAL_TOP, cx+CELL_W-2, CAL_TOP+CAL_H,
                           RED, DOT_PIXEL_2X2, DRAW_FILL_EMPTY);
      Paint_SelectImage(&paintB);
    }

    // 曜日ラベル
    Paint_SelectImage(isSun ? &paintR : &paintB);
    Paint_DrawString_EN(cx+4, CAL_TOP+4, dows[i], &Font20,
                        WHITE, isToday ? RED : BLACK);

    // 日付数字
    Paint_SelectImage(&paintB);
    sprintf(buf, "%2d", days[i]);
    Paint_DrawString_EN(cx+4, CAL_TOP+28, buf, &Font20,
                        WHITE, isToday ? RED : BLACK);

    // イベントドット
    if (hasEvent) {
      Paint_SelectImage(isToday ? &paintR : &paintB);
      Paint_DrawCircle(cx + CELL_W/2, CAL_TOP+CAL_H-8,
                       3, isToday?RED:BLACK, DOT_PIXEL_1X1, DRAW_FILL_FULL);
    }
  }

  // イベントリスト（3件）
  Paint_SelectImage(&paintB);
  for (int i = 0; i < evCount && i < 3; i++) {
    int ey = EVT_TOP + i * 28;
    // 左アクセントバー
    Paint_SelectImage(events[i].isToday ? &paintR : &paintB);
    Paint_DrawRectangle(COL_DIV+4, ey, COL_DIV+7, ey+22,
                        BLACK, DOT_PIXEL_1X1, DRAW_FILL_FULL);
    // 日時
    sprintf(buf, "%s %s", events[i].d, events[i].t);
    Paint_SelectImage(&paintB);
    Paint_DrawString_EN(COL_DIV+12, ey+2, buf, &Font20, WHITE, BLACK);
    // タイトル（ASCII時）
    Paint_DrawString_EN(COL_DIV+148, ey+2, events[i].n, &Font20, WHITE, BLACK);
  }

  // ── 転送・表示 ──────────────────────────────────────────
  EPD_5in79_Init();
  EPD_5in79_Display(imgBlack, imgRed);
  EPD_5in79_Sleep();   // 表示後すぐスリープ（焼き付き防止）
}
utils.cpp — getTimeStr() C++ 
#include <time.h>

void getTimeStr(char* buf) {
  time_t    now = time(nullptr);
  struct tm ti;
  localtime_r(&now, &ti);
  sprintf(buf, "%02d:%02d", ti.tm_hour, ti.tm_min);
}

void syncNTP() {
  configTzTime("JST-9", "ntp.nict.jp", "pool.ntp.org");
  struct tm ti;
  if (!getLocalTime(&ti, 5000)) {
    Serial.println("NTP sync failed");
  }
}
日本語文字列について： Waveshare の標準 GUI_Paint は ASCII フォントのみ。 日本語を描画するには ①u8g2 ライブラリのビットマップを流用する、 ②イベント名をローマ字・略称にする、③カスタムビットマップフォントを用意する、 のいずれかが必要です。Google Calendar 側でイベント名を短い英数字にしておくのが最も手軽です。
